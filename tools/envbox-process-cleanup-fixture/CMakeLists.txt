cmake_minimum_required(VERSION 3.20)
project(envbox-process-cleanup-fixture LANGUAGES CXX)
set(DETOURS_ROOT "D:/Tools/Detours" CACHE PATH "Built Microsoft Detours")
add_executable(envbox-process-cleanup-fixture main.cpp)
target_compile_features(envbox-process-cleanup-fixture PRIVATE cxx_std_17)
target_include_directories(envbox-process-cleanup-fixture PRIVATE "${DETOURS_ROOT}/include")
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
  target_link_libraries(envbox-process-cleanup-fixture PRIVATE "${DETOURS_ROOT}/lib.X64/detours.lib")
else()
  target_link_libraries(envbox-process-cleanup-fixture PRIVATE "${DETOURS_ROOT}/lib.X86/detours.lib")
endif()
add_library(cleanup-fault SHARED fault.cpp cleanup-fault.def)
target_compile_features(cleanup-fault PRIVATE cxx_std_17)
target_include_directories(cleanup-fault PRIVATE "${DETOURS_ROOT}/include")
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
  set_target_properties(cleanup-fault PROPERTIES OUTPUT_NAME cleanup-fault64)
  target_link_libraries(cleanup-fault PRIVATE "${DETOURS_ROOT}/lib.X64/detours.lib")
else()
  set_target_properties(cleanup-fault PROPERTIES OUTPUT_NAME cleanup-fault32)
  target_link_libraries(cleanup-fault PRIVATE "${DETOURS_ROOT}/lib.X86/detours.lib")
endif()
