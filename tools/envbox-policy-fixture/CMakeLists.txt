cmake_minimum_required(VERSION 3.24)
project(envbox_policy_fixture LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Windows MSVC is required")
endif()
add_executable(envbox-policy-fixture main.c ../../drivers/envbox-policy/policy.c ../../drivers/envbox-policy/network_snapshot.c)
target_include_directories(envbox-policy-fixture PRIVATE ../../drivers/envbox-policy)
target_compile_options(envbox-policy-fixture PRIVATE /W4 /WX /GS /TC)
target_link_libraries(envbox-policy-fixture PRIVATE advapi32)
enable_testing()
add_test(NAME policy-state-machine COMMAND envbox-policy-fixture)
add_executable(envbox-policy-cxx-linkage linkage.cpp ../../drivers/envbox-policy/policy.c)
target_include_directories(envbox-policy-cxx-linkage PRIVATE ../../drivers/envbox-policy)
target_compile_options(envbox-policy-cxx-linkage PRIVATE /W4 /WX)
add_test(NAME policy-cxx-linkage COMMAND envbox-policy-cxx-linkage)
