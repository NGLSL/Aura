cmake_minimum_required(VERSION 3.20)
project(envbox-doh-ffi-smoke LANGUAGES CXX)
if(NOT MSVC OR NOT IS_ABSOLUTE "${DOH_STATICLIB}")
  message(FATAL_ERROR "MSVC and absolute DOH_STATICLIB required")
endif()
add_library(doh-ffi-smoke SHARED bridge.cpp smoke.def)
target_include_directories(doh-ffi-smoke PRIVATE ../../crates/envbox-dns-doh/include)
target_link_libraries(doh-ffi-smoke PRIVATE "${DOH_STATICLIB}" ${DOH_NATIVE_LIBS})
target_link_options(doh-ffi-smoke PRIVATE ${DOH_NATIVE_OPTIONS})
add_executable(doh-ffi-host host.cpp)
target_link_libraries(doh-ffi-host PRIVATE ws2_32)
set_property(TARGET doh-ffi-smoke doh-ffi-host PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
