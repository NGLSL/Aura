cmake_minimum_required(VERSION 3.20)
project(envbox_doh_api_trap LANGUAGES CXX)
add_library(envbox-doh-api-trap SHARED api-trap.cpp)
target_compile_features(envbox-doh-api-trap PRIVATE cxx_std_17)
target_compile_definitions(envbox-doh-api-trap PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN)
set(DETOURS_ROOT "D:/Tools/Detours" CACHE PATH "Detours install")
target_include_directories(envbox-doh-api-trap PRIVATE "${DETOURS_ROOT}/include")
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
 target_link_libraries(envbox-doh-api-trap PRIVATE "${DETOURS_ROOT}/lib.X64/detours.lib")
else()
 target_link_libraries(envbox-doh-api-trap PRIVATE "${DETOURS_ROOT}/lib.X86/detours.lib")
 file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/exports.def" "EXPORTS\nDoHApiTrapInstall=_DoHApiTrapInstall@0\nDoHApiTrapSnapshot=_DoHApiTrapSnapshot@8\nDoHApiTrapSelfTest=_DoHApiTrapSelfTest@0\nDoHApiTrapAllowEndpoint=_DoHApiTrapAllowEndpoint@8\n")
 target_sources(envbox-doh-api-trap PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/exports.def")
endif()
target_link_libraries(envbox-doh-api-trap PRIVATE ws2_32 dnsapi winhttp crypt32)
add_executable(envbox-doh-api-trap-check trap-check.cpp)
target_compile_features(envbox-doh-api-trap-check PRIVATE cxx_std_17)
target_link_libraries(envbox-doh-api-trap-check PRIVATE ws2_32)
