cmake_minimum_required(VERSION 3.20)
project(envbox-startup-gate-fixture LANGUAGES CXX)
foreach(name IN ITEMS entry tls return)
  add_executable(envbox-startup-gate-${name} main.cpp)
  target_compile_options(envbox-startup-gate-${name} PRIVATE /GS- /Zl)
  target_link_options(envbox-startup-gate-${name} PRIVATE /NODEFAULTLIB /ENTRY:FixtureEntry /SUBSYSTEM:CONSOLE)
  target_link_libraries(envbox-startup-gate-${name} PRIVATE kernel32)
endforeach()
target_compile_definitions(envbox-startup-gate-tls PRIVATE FIXTURE_TLS=1)
target_compile_definitions(envbox-startup-gate-return PRIVATE FIXTURE_RETURN=1)

add_executable(envbox-startup-gate-console-crt crt_main.cpp)
add_executable(envbox-startup-gate-gui-crt WIN32 crt_main.cpp)
target_compile_definitions(envbox-startup-gate-gui-crt PRIVATE FIXTURE_GUI=1)
add_executable(envbox-startup-gate-gui main.cpp)
target_compile_options(envbox-startup-gate-gui PRIVATE /GS- /Zl)
target_link_options(envbox-startup-gate-gui PRIVATE /NODEFAULTLIB /ENTRY:FixtureEntry /SUBSYSTEM:WINDOWS)
target_link_libraries(envbox-startup-gate-gui PRIVATE kernel32)
